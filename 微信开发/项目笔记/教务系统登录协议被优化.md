## 首要分析

教务系统原始登录URL: http://jw.bkty.top:89/jsxsd/xk/LoginToXk
登录时携带参数 `service=http%3A%2F%2Fjw.bkty.top%3A89%2Fjsxsd%2Fsso.jsp`

更新后的登录URL: https://authserver.bkty.top/authserver/login
登录时携带参数 `service=https%3A%2F%2Fehall.bkty.top%2Flogin`

## 初步判断

经过多轮登录测试抓包分析后，我发现系统对POST请求时的参数进行了调整，看似对 `password` 和 `execution` 参数进行了加密处理。之前的登录只包含 `password`，对其进行Base64编码然后将 `password` 和 `username` 拼接即可，而现在明显变复杂了。

## 解决password加/解密

### 分析加/解密逻辑

经过断点调试分析后，我找到了 `password` 的加/解密逻辑，如下：

```js
function getAesString(n, f, c) {
    f = f.replace(/(^\s+)|(\s+$)/g, "");
    f = CryptoJS.enc.Utf8.parse(f);
    c = CryptoJS.enc.Utf8.parse(c);
    return CryptoJS.AES.encrypt(n, f, {
        iv: c,
        mode: CryptoJS.mode.CBC,
        padding: CryptoJS.pad.Pkcs7
    }).toString()
}
function encryptAES(n, f) {
    return f ? getAesString(randomString(64) + n, f, randomString(16)) : n
}
function encryptPassword(n, f) {
    try {
        return encryptAES(n, f)
    } catch (c) {}
    return n
}
var $aes_chars = "ABCDEFGHJKMNPQRSTWXYZabcdefhijkmnprstwxyz2345678"
  , aes_chars_len = $aes_chars.length;
function randomString(n) {
    var f = "";
    for (i = 0; i < n; i++)
        f += $aes_chars.charAt(Math.floor(Math.random() * aes_chars_len));
    return f
}
function decryptPassword(n, f) {
    var c = CryptoJS.enc.Utf8.parse(f)
      , q = CryptoJS.enc.Utf8.parse(randomString(16))
      , c = CryptoJS.AES.decrypt(n, c, {
        iv: q,
        mode: CryptoJS.mode.CBC,
        padding: CryptoJS.pad.Pkcs7
    });
    return CryptoJS.enc.Utf8.stringify(c).substring(64)
}
;
```

这个加密流程可以完整拆解为以下步骤：

1.核心函数与参数

- **`encryptPassword(n, f)`**：入口函数，明文 `n = "password"`，密钥 `f = "qzYOhusq5pdG0HpZ"`。
- **`encryptAES(n, f)`**：根据密钥 `f` 是否存在，决定是否调用 `getAesString`。
- **`getAesString(n, f, c)`**：真正的 AES 加密实现。

2.加密流程拆解

步骤 1：密钥与 IV 处理

```javascript
// 去除密钥首尾空白
f = f.replace(/(^\s*)|(\s*$)/g, "");
// 将密钥和IV从UTF-8字符串转为CryptoJS WordArray
f = CryptoJS.enc.Utf8.parse(f);
c = CryptoJS.enc.Utf8.parse(c);
```

- **密钥 (Key)**：`f = "qzYOhusq5pdG0HpZ"`，经 UTF-8 解析后作为 AES 密钥。
- **IV 向量**：`c` 由 `encryptAES` 传入，是一个 16 字节的随机字符串（`randomString(16)`）。

步骤 2：明文预处理

```javascript
// 当密钥f存在时，明文n会被包装
n = randomString(64) + n + randomString(16)
```

- 明文会被前后拼接随机字符串，增加破解难度。

步骤 3：AES 加密

```javascript
return CryptoJS.AES.encrypt(n, f, {
    iv: c,
    mode: CryptoJS.mode.CBC,
    padding: CryptoJS.pad.Pkcs7
}).toString()
```

- **算法**：AES
- **模式**：CBC（需要 IV 向量）
- **填充**：PKCS7
- **输出**：Base64 编码的字符串（`.toString()` 默认输出 Base64）

---

3.与你之前例子的对应关系

- **源数据**：`zhangsan123`
- **加密后数据**：`5DRD+admenZl9WWjS8p7uW68Mt35cc4nOFLx0cvBTQWW0uwVddzBr//yBYdkAWGHY+SyzITpKdRKettRtPfZyFcx/0HF/8x1Rpfcqxcgxn0=`
- **结论**：这个 Base64 字符串正是上述 AES-CBC-PKCS7 加密后，再经 Base64 编码的结果。

### 实现加密/解逻辑

```python
import base64  
import random  
from Crypto.Cipher import AES  
from Crypto.Util.Padding import pad, unpad  
from Crypto.Util.strxor import strxor  
  
# 前端限定的随机字符串字符集  
AES_CHARS = "ABCDEFGHJKMNPQRSTWXYZabcdefhijkmnprstwxyz2345678"  
AES_CHARS_LEN = len(AES_CHARS)  
  
  
def random_string(n):  
    """  
    完全复刻前端randomString函数  
    :param n: 生成字符串长度  
    :return: 限定字符集的随机字符串  
    """    f = ""  
    for _ in range(n):  
        # 随机取字符集中的字符（匹配前端Math.floor(Math.random() * aes_chars_len)）  
        index = random.randint(0, AES_CHARS_LEN - 1)  
        f += AES_CHARS[index]  
    return f  
  
  
def get_aes_string(n: str, f: str, c: str) -> str:  
    """  
    复刻前端getAesString函数  
    :param n: 待加密明文  
    :param f: 加密密钥  
    :param c: IV向量（16位字符串）  
    :return: Base64编码的AES加密结果  
    """    # 1. 去除密钥首尾空白（匹配前端f = f.replace(/(^\s+)|(\s+$)/g, "")）  
    f = f.strip()  
    # 2. 密钥/IV转为UTF-8字节（匹配前端CryptoJS.enc.Utf8.parse）  
    key = f.encode("utf-8")  
    iv = c.encode("utf-8")  
  
    # 3. 明文转UTF-8字节  
    plaintext_bytes = n.encode("utf-8")  
  
    # 4. AES-CBC加密 + PKCS7填充（匹配前端CryptoJS.AES.encrypt）  
    cipher = AES.new(key, AES.MODE_CBC, iv)  
    # PKCS7填充（AES块大小128位=16字节）  
    padded_data = pad(plaintext_bytes, AES.block_size, style="pkcs7")  
    encrypted_bytes = cipher.encrypt(padded_data)  
  
    # 5. Base64编码并转字符串（匹配前端.toString()）  
    return base64.b64encode(encrypted_bytes).decode("utf-8")  
  
  
def encrypt_aes(n: str, f: str) -> str:  
    """  
    复刻前端encryptAES函数  
    :param n: 原始明文  
    :param f: 加密密钥  
    :return: 加密结果（密钥为空则返回原明文）  
    """    if f:  
        # 拼接64位随机字符串 + 原始明文（前端逻辑：randomString(64) + n）  
        processed_plaintext = random_string(64) + n  
        # 生成16位随机IV向量  
        iv = random_string(16)  
        # 调用AES加密  
        return get_aes_string(processed_plaintext, f, iv)  
    else:  
        return n  
  
  
def encrypt_password(n: str, f: str) -> str:  
    """  
    复刻前端encryptPassword函数  
    :param n: 原始明文  
    :param f: 加密密钥  
    :return: 加密结果（异常则返回原明文）  
    """    try:  
        return encrypt_aes(n, f)  
    except Exception:  
        return n  
  
  
def decrypt_password(n: str, f: str) -> str:  
    """  
    复刻前端decryptPassword函数  
    :param n: 加密后的Base64字符串  
    :param f: 解密密钥（需和加密密钥一致）  
    :return: 解密后的原始明文（截取前64位随机字符串后内容）  
    """    try:  
        # 1. 密钥转UTF-8字节（匹配前端CryptoJS.enc.Utf8.parse(f)）  
        key = f.strip().encode("utf-8")  
        # 2. 生成16位随机IV（注意：前端解密时IV是随机生成的，这是个BUG！）  
        iv = random_string(16).encode("utf-8")  
  
        # 3. Base64解码加密数据（匹配前端CryptoJS.AES.decrypt）  
        encrypted_bytes = base64.b64decode(n)  
  
        # 4. AES-CBC解密 + PKCS7去填充  
        cipher = AES.new(key, AES.MODE_CBC, iv)  
        decrypted_bytes = cipher.decrypt(encrypted_bytes)  
        # 去除PKCS7填充  
        unpadded_data = unpad(decrypted_bytes, AES.block_size, style="pkcs7")  
        # 转字符串并截取前64位随机字符（匹配前端.substring(64)）  
        plaintext = unpadded_data.decode("utf-8")[64:]  
        return plaintext  
    except Exception as e:  
        # 解密失败返回原密文（模拟前端异常处理）  
        print(f"解密异常: {e}")  
        return n  
  
  
# ------------------- 测试示例 -------------------
if __name__ == "__main__":  
    # 测试参数（与前端一致）  
    original_text = "zhangsan123"  # 此处假设用户输入的密码是 zhangsan123
    aes_key = "qzYOhusq5pdG0HpZ"  
  
    # 1. 加密测试  
    encrypted_result = encrypt_password(original_text, aes_key)  
    print("加密结果：", encrypted_result)  
    print("加密结果长度：", len(encrypted_result))  
  
    # 2. 解密测试（注意：前端解密逻辑有BUG，IV随机生成，大概率解密失败！）  
    decrypted_result = decrypt_password(encrypted_result, aes_key)  
    print("解密结果：", decrypted_result)
```

## 解决execution参数

通过分析发现该参数并非是加密内容，`execution` 是服务器下发的登录票据，直接拿、直接提交，不需要你加密。所以只需对登录网站先 `GET` 一下，然后拿到动态得到的 `execution`，接着结合 `password` 和 `username` 字段直接对登录接口发起请求即可。获取 `execution` 的代码块如下：

```python
import requests  
from lxml import etree  
from urllib3.exceptions import InsecureRequestWarning  
  
# 禁用SSL警告（避免控制台报错）  
requests.packages.urllib3.disable_warnings(InsecureRequestWarning)  
  
  
def get_execution(url):  
    """  
    从指定URL提取execution值（通过XPath）  
    :param url: 登录页面URL  
    :return: execution值（提取失败返回空字符串）  
    """    try:  
        # 1. 发送GET请求（禁用SSL验证，解决证书问题）  
        response = requests.get(  
            url,  
            verify=False,  # 跳过SSL证书校验  
            headers={  
                "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"  
            },  
            timeout=10  # 超时保护  
        )  
        response.raise_for_status()  # 捕获HTTP错误（如404/500）  
        print(response.text)  
  
        # 2. 解析HTML（自动处理编码）  
        html = etree.HTML(response.text)  
  
        # 3. 通过XPath提取execution值  
        # 你的XPath路径：
	        # //*[@id="qrLoginForm"]/input[5]  
			# /html/body/div[6]/div/div/form/input[5]
        execution_ele = html.xpath(  
            '//*[@id="qrLoginForm"]/input[5]')  
  
        # 4. 提取value属性值  
        print(execution_ele)  
        if execution_ele and len(execution_ele) > 0:  
            execution = execution_ele[0].get("value", "")  
            print(f"成功提取execution：{execution}")  
            return execution  
        else:  
            print("未找到execution元素（XPath可能有误）")  
            return ""  
  
    except Exception as e:  
        print(f"提取execution失败：{str(e)}")  
        return ""  
  
  
# ------------------- 测试使用 -------------------
if __name__ == "__main__":  
    # 目标登录页面URL  
    login_url = "https://authserver.bkty.top/authserver/login"  
  
    # 提取execution  
    execution = get_execution(login_url)  
  
    # 后续可结合之前的AES加密逻辑，构造POST请求  
    if execution:  
        # 这里放入你的登录POST逻辑  
        print("开始构造登录请求...")  
    else:  
        print("execution提取失败，无法继续登录")  
  
```
